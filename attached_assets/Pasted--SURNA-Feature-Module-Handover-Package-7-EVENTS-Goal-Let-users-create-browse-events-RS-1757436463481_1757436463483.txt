SURNA – Feature Module Handover (Package #7: EVENTS)

Goal

Let users create/browse events, RSVP, and (optionally) issue free tickets. Clean module, cursor pagination, auth where needed.


---

DB (migrations)

-- events created by users/teams
CREATE TABLE IF NOT EXISTS events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  creator_id UUID NOT NULL,
  title TEXT NOT NULL,
  description TEXT DEFAULT '',
  starts_at TIMESTAMPTZ NOT NULL,
  ends_at TIMESTAMPTZ NOT NULL,
  location TEXT DEFAULT '',
  visibility TEXT NOT NULL DEFAULT 'public',     -- 'public' | 'private' | 'unlisted'
  capacity INT NULL,                             -- null = unlimited
  cover_media_id UUID NULL,                      -- link to media
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_events_time ON events (starts_at DESC, id DESC);
CREATE INDEX IF NOT EXISTS idx_events_visibility_time ON events (visibility, starts_at DESC, id DESC);

-- RSVPs, one per user per event
CREATE TABLE IF NOT EXISTS event_rsvps (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
  user_id UUID NOT NULL,
  status TEXT NOT NULL DEFAULT 'going',          -- 'going' | 'interested' | 'not_going'
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE(event_id, user_id)
);
CREATE INDEX IF NOT EXISTS idx_rsvps_event ON event_rsvps (event_id);

-- Optional free tickets (issued upon RSVP=going)
CREATE TABLE IF NOT EXISTS event_tickets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_id UUID NOT NULL,
  user_id UUID NOT NULL,
  code TEXT NOT NULL UNIQUE,
  issued_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

---

“Sayings” (rules)

Create/Update/Delete require auth; only creator can edit/delete.

List returns public (and unlisted if the user has the direct id).

Cursor pagination by (starts_at,id) for future events; filter time windows.

RSVP enforces capacity if set (count GOING).

On RSVP=going, optionally issue ticket with unique code.

Times are ISO strings (UTC).

import { z } from "zod";

export const CreateEvent = z.object({
  title: z.string().min(1).max(140),
  description: z.string().max(4000).optional(),
  startsAt: z.string().datetime(),
  endsAt: z.string().datetime(),
  location: z.string().max(255).optional(),
  visibility: z.enum(["public","private","unlisted"]).default("public"),
  capacity: z.number().int().min(1).optional(),
  coverMediaId: z.string().uuid().optional(),
});

export const UpdateEvent = CreateEvent.partial();

export const ListQuery = z.object({
  from: z.string().datetime().optional(),           // default: now
  to: z.string().datetime().optional(),             // optional upper bound
  q: z.string().optional(),
  cursorStartsAt: z.string().datetime().optional(),
  cursorId: z.string().uuid().optional(),
  limit: z.coerce.number().min(1).max(50).default(20),
});

export const RSVPBody = z.object({
  status: z.enum(["going","interested","not_going"]).default("going"),
  issueTicket: z.boolean().default(true),
});

events.repo.ts

import { db } from "../../core/db";
import { sql } from "drizzle-orm";

export async function insertEvent(creatorId: string, e: any) {
  const q = await db.execute(sql`
    INSERT INTO events (creator_id, title, description, starts_at, ends_at, location, visibility, capacity, cover_media_id)
    VALUES (${creatorId}, ${e.title}, ${e.description ?? ''}, ${e.startsAt}, ${e.endsAt},
            ${e.location ?? ''}, ${e.visibility ?? 'public'}, ${e.capacity ?? null}, ${e.coverMediaId ?? null})
    RETURNING *;
  `);
  return q.rows[0];
}

export async function updateEvent(creatorId: string, id: string, e: any) {
  const q = await db.execute(sql`
    UPDATE events SET
      title = COALESCE(${e.title}, title),
      description = COALESCE(${e.description}, description),
      starts_at = COALESCE(${e.startsAt}, starts_at),
      ends_at = COALESCE(${e.endsAt}, ends_at),
      location = COALESCE(${e.location}, location),
      visibility = COALESCE(${e.visibility}, visibility),
      capacity = COALESCE(${e.capacity}, capacity),
      cover_media_id = COALESCE(${e.coverMediaId}, cover_media_id)
    WHERE id=${id} AND creator_id=${creatorId}
    RETURNING *;
  `);
  return q.rows[0] ?? null;
}

export async function deleteEvent(creatorId: string, id: string) {
  const r = await db.execute(sql`DELETE FROM events WHERE id=${id} AND creator_id=${creatorId} RETURNING id;`);
  return r.rows[0]?.id ?? null;
}

export async function getEvent(id: string) {
  const q = await db.execute(sql`SELECT * FROM events WHERE id=${id} LIMIT 1;`);
  return q.rows[0] ?? null;
}

export async function countGoing(eventId: string) {
  const q = await db.execute(sql`SELECT COUNT(*)::int AS c FROM event_rsvps WHERE event_id=${eventId} AND status='going';`);
  return q.rows[0]?.c ?? 0;
}

export async function listPublic(qs: {from?: string,to?: string,q?: string,cursorStartsAt?:string,cursorId?:string,limit:number}) {
  const now = qs.from ?? new Date().toISOString();
  const timeWhere = qs.to
    ? sql`AND starts_at BETWEEN ${now}::timestamptz AND ${qs.to}::timestamptz`
    : sql`AND starts_at >= ${now}::timestamptz`;
  const search = qs.q
    ? sql`AND (title ILIKE '%'||${qs.q}||'%' OR description ILIKE '%'||${qs.q}||'%')`
    : sql``;
  const cursor = (qs.cursorStartsAt && qs.cursorId)
    ? sql`AND (starts_at, id) < (${qs.cursorStartsAt}::timestamptz, ${qs.cursorId}::uuid)`
    : sql``;

  const q = await db.execute(sql`
    SELECT * FROM events
    WHERE visibility='public'
    ${timeWhere} ${search} ${cursor}
    ORDER BY starts_at DESC, id DESC
    LIMIT ${qs.limit};
  `);
  return q.rows;
}

export async function upsertRSVP(eventId: string, userId: string, status: string) {
  const q = await db.execute(sql`
    INSERT INTO event_rsvps (event_id, user_id, status)
    VALUES (${eventId}, ${userId}, ${status})
    ON CONFLICT (event_id, user_id) DO UPDATE SET status=EXCLUDED.status
    RETURNING *;
  `);
  return q.rows[0];
}

export async function issueTicket(eventId: string, userId: string) {
  const code = `TKT-${Math.random().toString(36).slice(2,10).toUpperCase()}`;
  const q = await db.execute(sql`
    INSERT INTO event_tickets (event_id, user_id, code)
    VALUES (${eventId}, ${userId}, ${code})
    RETURNING *;
  `);
  return q.rows[0];
}

export async function listMyRSVPs(userId: string) {
  const q = await db.execute(sql`
    SELECT r.*, e.title, e.starts_at, e.location
    FROM event_rsvps r JOIN events e ON e.id=r.event_id
    WHERE r.user_id=${userId}
    ORDER BY e.starts_at DESC;
  `);
  return q.rows;
}

events.service.ts

import * as repo from "./events.repo";

export async function createEvent(creatorId: string, e: any) {
  if (new Date(e.endsAt) <= new Date(e.startsAt)) throw new Error("END_BEFORE_START");
  return await repo.insertEvent(creatorId, e);
}

export async function editEvent(creatorId: string, id: string, e: any) {
  if (e.startsAt && e.endsAt && new Date(e.endsAt) <= new Date(e.startsAt)) throw new Error("END_BEFORE_START");
  return await repo.updateEvent(creatorId, id, e);
}

export async function rsvp(eventId: string, userId: string, status: "going"|"interested"|"not_going", issueTicket: boolean) {
  // capacity check if going
  if (status === "going") {
    const current = await repo.countGoing(eventId);
    const ev = await repo.getEvent(eventId);
    if (!ev) throw new Error("EVENT_NOT_FOUND");
    if (ev.capacity && current >= ev.capacity) throw new Error("EVENT_FULL");
  }
  const row = await repo.upsertRSVP(eventId, userId, status);
  let ticket = null;
  if (status === "going" && issueTicket) ticket = await repo.issueTicket(eventId, userId);
  return { rsvp: row, ticket };
}

events.router.ts

import { Router } from "express";
import { z } from "zod";
import { CreateEvent, UpdateEvent, ListQuery, RSVPBody } from "./events.validation";
import { createEvent, editEvent, rsvp } from "./events.service";
import * as repo from "./events.repo";

export const eventsRouter = Router();

// PUBLIC: list events (future window)
eventsRouter.get("/", async (req, res, next) => {
  try {
    const q = ListQuery.parse(req.query);
    const rows = await repo.listPublic(q);
    const nextCursor = rows.length ? { cursorStartsAt: rows[rows.length-1].starts_at, cursorId: rows[rows.length-1].id } : null;
    res.json({ items: rows, nextCursor });
  } catch (e) { next(e); }
});

// PUBLIC: get one event (public/unlisted if you know id)
eventsRouter.get("/:id", async (req, res, next) => {
  try {
    const id = z.string().uuid().parse(req.params.id);
    const ev = await repo.getEvent(id);
    if (!ev) return res.status(404).json({ error: "NOT_FOUND" });
    if (ev.visibility === "private") return res.status(403).json({ error: "FORBIDDEN" });
    res.json(ev);
  } catch (e) { next(e); }
});

// AUTH: create
eventsRouter.post("/", async (req: any, res, next) => {
  try {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHORIZED" });
    const body = CreateEvent.parse(req.body);
    const ev = await createEvent(req.user.id, body);
    res.status(201).json(ev);
  } catch (e) { next(e); }
});

// AUTH: update (creator only)
eventsRouter.patch("/:id", async (req: any, res, next) => {
  try {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHORIZED" });
    const id = z.string().uuid().parse(req.params.id);
    const body = UpdateEvent.parse(req.body);
    const ev = await editEvent(req.user.id, id, body);
    if (!ev) return res.status(404).json({ error: "NOT_FOUND" });
    res.json(ev);
  } catch (e) { next(e); }
});

// AUTH: delete (creator only)
eventsRouter.delete("/:id", async (req: any, res, next) => {
  try {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHORIZED" });
    const id = z.string().uuid().parse(req.params.id);
    const ok = await repo.deleteEvent(req.user.id, id);
    if (!ok) return res.status(404).json({ error: "NOT_FOUND" });
    res.status(204).end();
  } catch (e) { next(e); }
});

// AUTH: RSVP
eventsRouter.post("/:id/rsvp", async (req: any, res, next) => {
  try {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHORIZED" });
    const id = z.string().uuid().parse(req.params.id);
    const { status, issueTicket } = RSVPBody.parse(req.body);
    const out = await rsvp(id, req.user.id, status, issueTicket);
    res.status(201).json(out);
  } catch (e) { next(e); }
});

// AUTH: my RSVPs
eventsRouter.get("/me/rsvps", async (req: any, res, next) => {
  try {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHORIZED" });
    const rows = await repo.listMyRSVPs(req.user.id);
    res.json({ items: rows });
  } catch (e) { next(e); }
});
