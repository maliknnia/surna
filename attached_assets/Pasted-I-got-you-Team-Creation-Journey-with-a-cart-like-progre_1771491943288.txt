Cart Logic

```javascript
const teamBuilder = {
  steps: ['sport', 'venue', 'players', 'gear', 'boost'],
  currentStep: 0,
  data: {
    sport: null,
    venue: null,
    players: [],
    gear: [],
    boosts: []
  },
  
  canLaunch() {
    // Required: sport + venue
    return this.data.sport && this.data.venue;
  },
  
  skipStep(step) {
    // Mark as skipped, continue
    this.data[step] = 'skipped';
    this.next();
  },
  
  calculatePoints() {
    let points = 500; // Base
    if (this.data.venue) points += 100;
    if (this.data.players.length >= 5) points += 200;
    points += this.data.gear.length * 50;
    return points;
  }
};
```
