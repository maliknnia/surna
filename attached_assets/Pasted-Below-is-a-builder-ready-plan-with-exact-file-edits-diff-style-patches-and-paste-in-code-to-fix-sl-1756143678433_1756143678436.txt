2) Postgres/Drizzle connection pooling (server/db.ts)

Goal: ensure minimal connection overhead and add query logging for slow queries.

Edit: server/db.ts

// server/db.ts
import { drizzle } from 'drizzle-orm/node-postgres';
import { Pool } from 'pg';

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 10, // tune per deployment size (10–50)
  idleTimeoutMillis: 30000,
  connectionTimeoutMillis: 10000,
});

export const db = drizzle(pool, {
  logger: {
    logQuery: (query, params) => {
      // lightweight slow-query log (threshold ~150ms)
      const start = Date.now();
      return {
        then: (onfulfilled: any, onrejected: any) =>
          Promise.resolve()
            .then(() => onfulfilled?.(null))
            .catch(onrejected)
            .finally(() => {
              const ms = Date.now() - start;
              if (ms > 150) console.warn(`[DB SLOW ${ms}ms]`, query.slice(0, 120), params?.slice?.(0, 5));
            }),
      } as any;
    },
  },
});

> If you’re already using neon-http, you can keep it; the pg pool above is a safe default and generally faster for Node servers.

7) API pagination — cursor-based for feed & messages (server/routes.ts)

Edit: implement cursor pagination using createdAt cursors. Ensure idx_*_created_at indexes exist (from Step 1).

// server/routes.ts (excerpt)
import express from 'express';
import { db } from './db';

const router = express.Router();

router.get('/api/feed', async (req, res) => {
  const limit = Math.min(Number(req.query.limit) || 20, 50);
  const cursor = req.query.cursor ? new Date(String(req.query.cursor)) : null;

  const rows = await db.execute(
    `SELECT * FROM posts
     WHERE ($1::timestamptz IS NULL OR "createdAt" < $1)
     ORDER BY "createdAt" DESC
     LIMIT $2`,
    [cursor, limit + 1]
  );

  const next = rows.length > limit ? rows[limit - 1]?.createdAt : null;
  res.json({ items: rows.slice(0, limit), nextCursor: next });
});

router.get('/api/messages', async (req, res) => {
  const { conversationId } = req.query as { conversationId: string };
  const limit = Math.min(Number(req.query.limit) || 30, 100);
  const cursor = req.query.cursor ? new Date(String(req.query.cursor)) : null;

  const rows = await db.execute(
    `SELECT * FROM messages
     WHERE "conversationId" = $1
       AND ($2::timestamptz IS NULL OR "createdAt" < $2)
     ORDER BY "createdAt" DESC
     LIMIT $3`,
    [conversationId, cursor, limit + 1]
  );

  const next = rows.length > limit ? rows[limit - 1]?.createdAt : null;
  res.json({ items: rows.slice(0, limit), nextCursor: next });
});

export default router;

9) Client media compression before upload

Create: client/src/lib/media.ts

// client/src/lib/media.ts
export async function compressImage(file: File, maxW = 1600, quality = 0.82) {
  const img = document.createElement('img');
  img.src = URL.createObjectURL(file);
  await img.decode();

  const scale = Math.min(1, maxW / img.naturalWidth);
  const canvas = document.createElement('canvas');
  canvas.width = Math.round(img.naturalWidth * scale);
  canvas.height = Math.round(img.naturalHeight * scale);
  const ctx = canvas.getContext('2d')!;
  ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
  const blob = await new Promise<Blob | null>(r => canvas.toBlob(r, 'image/jpeg', quality));
  if (!blob) return file;
  return new File([blob], file.name.replace(/\.[^.]+$/, '.jpg'), { type: 'image/jpeg' });
}

Use it in CreatePost.tsx/uploaders before sending to server.

Stage 2 — Performance Scaling

11) Redis cache wrapper (server/storage.ts)

Goal: cache hot list endpoints (feed, coaches, products) for ~15–30s.

// server/storage.ts (excerpt)
import Redis from 'ioredis';
const redis = process.env.REDIS_URL ? new Redis(process.env.REDIS_URL) : null;

export async function cacheGet<T>(key: string): Promise<T | null> {
  if (!redis) return null;
  const raw = await redis.get(key);
  return raw ? JSON.parse(raw) as T : null;
}

export async function cacheSet(key: string, val: any, ttlSeconds = 30) {
  if (!redis) return;
  await redis.set(key, JSON.stringify(val), 'EX', ttlSeconds);
}

Use it inside feed route:

router.get('/api/feed', async (req, res) => {
  const limit = Math.min(Number(req.query.limit) || 20, 50);
  const cursor = req.query.cursor ? new Date(String(req.query.cursor)) : null;
  const k = `feed:${limit}:${cursor?.toISOString() ?? 'null'}`;

  const hit = await cacheGet(k);
  if (hit) return res.json(hit);

  // ... run DB query as in Stage 1
  const payload = { items: rows.slice(0, limit), nextCursor: next };
  await cacheSet(k, payload, 20);
  res.json(payload);
});

import { createServer } from 'http';

const io = new Server(httpServer, { cors: { origin: '*', credentials: true } });

if (process.env.REDIS_URL) {
  const pubClient = createClient({ url: process.env.REDIS_URL });
  const subClient = pubClient.duplicate();
  await pubClient.connect();
  await subClient.connect();
  io.adapter(createAdapter(pubClient, subClient));
}

io.on('connection', (socket) => {
  // existing message handlers
});

httpServer.listen(process.env.PORT || 3000);
