1) Database – migrations (all at once)

-- 1:1 conversations (already from DM; keep)
CREATE TABLE IF NOT EXISTS dm_conversations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_a UUID NOT NULL,
  user_b UUID NOT NULL,
  last_message_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (LEAST(user_a,user_b), GREATEST(user_a,user_b))
);
CREATE INDEX IF NOT EXISTS idx_dm_conv_last ON dm_conversations (last_message_at DESC);

-- 1:1 messages (text | audio)
CREATE TABLE IF NOT EXISTS dm_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  conversation_id UUID NOT NULL REFERENCES dm_conversations(id) ON DELETE CASCADE,
  sender_id UUID NOT NULL,
  kind TEXT NOT NULL DEFAULT 'text',     -- 'text' | 'audio'
  body TEXT NOT NULL DEFAULT '',
  media_id UUID NULL,                    -- audio -> media.id
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_dm_msg_conv_time ON dm_messages (conversation_id, created_at DESC, id DESC);

-- Read receipts for 1:1
CREATE TABLE IF NOT EXISTS dm_reads (
  conversation_id UUID NOT NULL REFERENCES dm_conversations(id) ON DELETE CASCADE,
  user_id UUID NOT NULL,
  last_read_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (conversation_id, user_id)
);

-- GROUPS
CREATE TABLE IF NOT EXISTS group_conversations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id UUID NOT NULL,
  name TEXT NOT NULL,
  description TEXT DEFAULT '',
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- membership roles: 'owner' | 'admin' | 'member'
CREATE TABLE IF NOT EXISTS group_members (
  group_id UUID NOT NULL REFERENCES group_conversations(id) ON DELETE CASCADE,
  user_id UUID NOT NULL,
  role TEXT NOT NULL DEFAULT 'member',
  joined_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (group_id, user_id)
);
CREATE INDEX IF NOT EXISTS idx_group_members_user ON group_members(user_id);

-- group messages (text | audio)
CREATE TABLE IF NOT EXISTS group_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  group_id UUID NOT NULL REFERENCES group_conversations(id) ON DELETE CASCADE,
  sender_id UUID NOT NULL,
  kind TEXT NOT NULL DEFAULT 'text',     -- 'text' | 'audio'
  body TEXT NOT NULL DEFAULT '',
  media_id UUID NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_group_msg_time ON group_messages (group_id, created_at DESC, id DESC);

-- read receipts per user per group (last seen timestamp)
CREATE TABLE IF NOT EXISTS group_reads (
  group_id UUID NOT NULL REFERENCES group_conversations(id) ON DELETE CASCADE,
  user_id UUID NOT NULL,
  last_read_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (group_id, user_id)
);

-- invites (owner/admin invites someone)
CREATE TABLE IF NOT EXISTS group_invites (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  group_id UUID NOT NULL REFERENCES group_conversations(id) ON DELETE CASCADE,
  inviter_id UUID NOT NULL,
  invitee_id UUID NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending',  -- 'pending'|'accepted'|'declined'|'canceled'
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (group_id, invitee_id)
);

-- join requests (user requests to join)
CREATE TABLE IF NOT EXISTS group_join_requests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  group_id UUID NOT NULL REFERENCES group_conversations(id) ON DELETE CASCADE,
  requester_id UUID NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending',  -- 'pending'|'approved'|'denied'
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (group_id, requester_id)
);

-- Messenger settings per user
CREATE TABLE IF NOT EXISTS messenger_settings (
  user_id UUID PRIMARY KEY,
  allow_message_requests BOOLEAN NOT NULL DEFAULT TRUE,
  call_permission TEXT NOT NULL DEFAULT 'everyone', -- 'everyone'|'following'|'none'
  read_receipts BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- (Optional) audio metadata in media
ALTER TABLE media ADD COLUMN IF NOT EXISTS duration_ms INT NULL;
ALTER TABLE media ADD COLUMN IF NOT EXISTS waveform JSONB NULL;

3) “Sayings” (rules to follow)

Rooms

1:1: dm:<conversationId>

Group: group:<groupId>

Per-user: user:<userId> (already in your IO)


Permissions

Only conversation participants can read 1:1 thread.

Group roles: owner can transfer ownership, add/remove admins/members; admin can invite/approve, remove members (not owner); member can post and leave.

Settings: respect call_permission & allow_message_requests.


Pagination

All message histories use cursor (created_at, id).


Voice notes

Use Media Pkg to upload audio (audio/webm|ogg|mpeg), then send message with mediaId.


Call signaling

Server only forwards: call:invite, call:offer, call:answer, call:candidate, call:end, call:reject.

Use STUN (free) + TURN (recommended) via env.

4) API – endpoints (compact)

4.1 1:1

POST /api/messenger/dm/conversations { peerId } → { id } (ensure/upsert)

GET /api/messenger/dm/conversations → list my DM threads (sorted by last message)

GET /api/messenger/dm/messages?conversationId=&cursorCreatedAt=&cursorId=&limit= → { items, nextCursor }

POST /api/messenger/dm/messages { conversationId? | peerId?, body? | mediaId? } → creates text or audio

POST /api/messenger/dm/read { conversationId } → sets read; emits dm:read


4.2 Groups

POST /api/messenger/groups { name, description? } → create + owner membership

PATCH /api/messenger/groups/:groupId { name?, description? } (owner/admin)

DELETE /api/messenger/groups/:groupId (owner)

GET /api/messenger/groups → groups I’m in

GET /api/messenger/groups/:groupId/messages?cursorCreatedAt=&cursorId=&limit= → history

POST /api/messenger/groups/:groupId/messages { body? | mediaId? } → text/audio

POST /api/messenger/groups/:groupId/read → set read


Membership

POST /api/messenger/groups/:groupId/members { userId, role? } (owner/admin add)

DELETE /api/messenger/groups/:groupId/members/:userId (owner/admin remove or self-leave)

PATCH /api/messenger/groups/:groupId/members/:userId { role } (owner → promote/demote)


Invites / Requests

POST /api/messenger/groups/:groupId/invites { inviteeId } (owner/admin)

POST /api/messenger/groups/:groupId/invites/:inviteeId/accept (invitee)

POST /api/messenger/groups/:groupId/invites/:inviteeId/decline (invitee)

POST /api/messenger/groups/:groupId/requests (user asks to join)

POST /api/messenger/groups/:groupId/requests/:requestId/approve (owner/admin)

POST /api/messenger/groups/:groupId/requests/:requestId/deny (owner/admin)


4.3 Settings

GET  /api/messenger/settings/me

PATCH /api/messenger/settings/me { allow_message_requests?, call_permission?, read_receipts? }


> All routes require JWT (except future public invite deep-links, which you can add later).

5) Socket.IO – events to add

Server (added to your io.on("connection")):

io.on("connection", (socket) => {
  const me = (socket as any).user.id;

  // Rooms
  socket.on("dm:join", ({ conversationId }) => socket.join(`dm:${conversationId}`));
  socket.on("group:join", ({ groupId }) => socket.join(`group:${groupId}`));
  socket.on("group:leave", ({ groupId }) => socket.leave(`group:${groupId}`));

  // 1:1 realtime
  // Emit "dm:new" on message creation (server service already does this)
  // Emit "dm:read" on read receipt (server service already does this)

  // Group realtime (server emits on post/add/remove/read)
  // "group:new" { groupId, message }
  // "group:read" { groupId, userId, at }
  // "group:member:add" | "group:member:remove" | "group:member:role"

  // Call signaling (1:1; optionally allow inside a group by picking a peer)
  socket.on("call:invite", ({ toUserId, conversationId, media }) => {
    io.to(`user:${toUserId}`).emit("call:ring", { fromUserId: me, conversationId, media, at: Date.now() });
  });
  socket.on("call:reject", ({ toUserId, conversationId }) => {
    io.to(`user:${toUserId}`).emit("call:rejected", { fromUserId: me, conversationId });
  });
  socket.on("call:offer", (p) => io.to(`user:${p.toUserId}`).emit("call:offer", { ...p, fromUserId: me }));
  socket.on("call:answer", (p) => io.to(`user:${p.toUserId}`).emit("call:answer", { ...p, fromUserId: me }));
  socket.on("call:candidate", (p) => io.to(`user:${p.toUserId}`).emit("call:candidate", { ...p, fromUserId: me }));
  socket.on("call:end", ({ toUserId, conversationId }) => {
    io.to(`user:${toUserId}`).emit("call:end", { fromUserId: me, conversationId });
  });
});
